function Pill({ label, colors }) {
  const c = colors || { bg: "rgba(200,214,229,0.1)", text: "#8899AA", border: "rgba(200,214,229,0.2)" };
  return (
    <span style={{
      background: c.bg, color: c.text, border: `1px solid ${c.border}`,
      padding: "2px 10px", borderRadius: 20, fontSize: 11, fontWeight: 600,
      letterSpacing: "0.08em", textTransform: "uppercase", whiteSpace: "nowrap"
    }}>{label}</span>
  );
}

function StatCard({ label, value, sub, accent }) {
  return (
    <div style={{
      background: "#1C2433", border: "1px solid rgba(200,214,229,0.08)",
      borderRadius: 10, padding: "24px 22px",

      {sub && <div style={{ fontSize: 12, color: "#6B7A94", marginTop: 6 }}>{sub}</div>}
    </div>
  );
}

            <div style={{ fontSize: 12, color: "#6B7A94", marginBottom: 28 }}>Vendor: {AGREEMENT.vendor} · v{AGREEMENT.version} · Effective {AGREEMENT.effectiveDate}</div>

            <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fit, minmax(180px, 1fr))", gap: 16, marginBottom: 32 }}>
              <StatCard label="Referral Reward" value={`${AGREEMENT.referralPercent}%`} sub="Of confirmed job value" accent="#F5B731" />
              <StatCard label="Closing Bonus" value={`R${AGREEMENT.closingFee}`} sub="Fixed, on job completion" accent="#F5B731" />
              <StatCard label="Agreement Signed" value={signedConnectors} sub={`of ${totalConnectors} connectors`} accent="#22C55E" />
            </div>

            <div style={{ background: "#1C2433", border: "1px solid rgba(200,214,229,0.08)", borderRadius: 10, overflow: "hidden", marginBottom: 32 }}>
              <div style={{ padding: "20px 24px", borderBottom: "1px solid rgba(200,214,229,0.06)" }}>
                <div style={{ fontSize: 11, letterSpacing: "0.2em", textTransform: "uppercase", color: "#6B7A94" }}>Signature Status</div>
              </div>
              {data.connectors.map(c => (
                <div key={c.id} style={{ padding: "14px 24px", borderBottom: "1px solid rgba(200,214,229,0.04)", display: "flex", alignItems: "center", gap: 16, flexWrap: "wrap" }}>
                  <div style={{ flex: 1, minWidth: 160 }}>
                    <div style={{ fontWeight: 600, color: "#EEF2F8" }}>{c.name}</div>
                    <div style={{ fontSize: 12, color: "#6B7A94" }}>{c.area}</div>
                  </div>
                  {c.agreementSigned ? (
                    <Pill label={`Signed ${c.agreementSignedDate || ""}`} colors={{ bg: "rgba(34,197,94,0.12)", text: "#4ADE80", border: "rgba(34,197,94,0.3)" }} />
                  ) : (
                    <Pill label="Not Signed" colors={{ bg: "rgba(239,68,68,0.1)", text: "#F87171", border: "rgba(239,68,68,0.25)" }} />
                  )}
                </div>
              ))}
            </div>

            <div style={{ background: "rgba(26,111,255,0.06)", border: "1px solid rgba(26,111,255,0.2)", borderRadius: 8, padding: "20px" }}>
              <div style={{ fontSize: 11, letterSpacing: "0.2em", textTransform: "uppercase", color: "#1A6FFF", marginBottom: 12 }}>Terms Summary</div>
              <ul style={{ paddingLeft: 18, color: "#C8D6E5", fontSize: 13, lineHeight: 1.9 }}>
                <li>Referral reward: {AGREEMENT.referralPercent}% of total job value, paid once the client's deposit clears and the job is confirmed.</li>
                <li>Closing bonus: R{AGREEMENT.closingFee} fixed, paid on top of the referral reward once the installation is completed and signed off.</li>
                <li>No fees, no joining costs, no obligation to sell.</li>
                <li>Full terms: <span style={{ color: "#F5B731" }}>primeturf.co.za/partner-agreement.html</span></li>
              </ul>
            </div>
          </div>
        )}

        {/* ── FLOW GUIDE ── */}
        {view === "flow" && (
          <div style={{ maxWidth: 680 }}>
            <div style={{ fontSize: 11, letterSpacing: "0.2em", textTransform: "uppercase", color: "#1A6FFF", marginBottom: 8 }}>Quick Reference</div>
            <h2 style={{ fontFamily: "monospace", fontSize: 24, fontWeight: 700, marginBottom: 28 }}>Tag & Routing Guide</h2>

            {[
              { tag: "PRIORITY", desc: "Active, fast-responding, proven connectors. Send all opportunities first.", action: "First-access list · All opportunities · Immediate" },
              { tag: "ACTIVE", desc: "Engaged, replied to flow, confirmed ready to connect.", action: "Second wave · Within 2h of opportunity drop" },
              { tag: "REFERRER", desc: "Identified as connector type. May not have converted yet.", action: "Included in opportunity broadcasts" },
              { tag: "SUPPLIER", desc: "Service provider or installer. Receives matched leads only.", action: "Direct matching — not broadcast" },
              { tag: "EXPLORER", desc: "Joined but hasn't committed. Needs more nurture.", action: "Third wave · 24h delay · Day 6 re-engagement" },
              { tag: "COLD", desc: "No reply in 72h. Send re-engagement nudge only.", action: "Day 6 nudge only · No opportunities" },
              { tag: "EXITED", desc: "Opted out. Archive only — do not broadcast.", action: "No broadcasts · Keep in archive" },
            ].map(item => (
              <div key={item.tag} style={{ background: "#1C2433", border: "1px solid rgba(200,214,229,0.07)", borderRadius: 8, padding: "18px 20px", marginBottom: 10 }}>
                <div style={{ display: "flex", alignItems: "center", gap: 12, marginBottom: 8 }}>
                  <Pill label={item.tag} colors={TAG_COLORS[item.tag]} />
                </div>
                <div style={{ fontSize: 13, color: "#C8D6E5", marginBottom: 6 }}>{item.desc}</div>
                <div style={{ fontSize: 12, color: "#6B7A94", fontFamily: "monospace" }}>→ {item.action}</div>
              </div>
            ))}

            <div style={{ background: "rgba(26,111,255,0.06)", border: "1px solid rgba(26,111,255,0.2)", borderRadius: 8, padding: "20px", marginTop: 24 }}>
              <div style={{ fontSize: 11, letterSpacing: "0.2em", textTransform: "uppercase", color: "#1A6FFF", marginBottom: 12 }}>Entry Keywords</div>
              {[["CONNECT / TURF / JOIN","Entry trigger → auto-reply + segment question"],["A","Tags REFERRER → sends Referrer reply"],["B","Tags SUPPLIER → sends Supplier reply"],["C","Tags EXPLORER → sends Explorer reply"],["READY","Tags ACTIVE → priority list"],["ACTIVE","Tags PRIORITY → first-access list"],["STAY","Re-activates COLD contact"],["OUT","Tags EXITED → clean removal"]].map(([kw, action]) => (
                <div key={kw} style={{ display: "flex", gap: 16, padding: "8px 0", borderBottom: "1px solid rgba(200,214,229,0.05)" }}>
                  <span style={{ fontFamily: "monospace", fontSize: 12, color: "#F5B731", minWidth: 140, flexShrink: 0 }}>{kw}</span>
                  <span style={{ fontSize: 12, color: "#6B7A94" }}>{action}</span>
                </div>
              ))}
            </div>
          </div>
        )}
      </div>

      {/* ── ADD CONNECTOR MODAL ── */}
      {showAddConnector && (
        <Modal title="Add Connector" onClose={() => setShowAddConnector(false)}>
          <Field label="Full Name *"><input style={inputStyle} value={newConnector.name} onChange={e => setNewConnector(p => ({...p, name: e.target.value}))} placeholder="e.g. Thabo Mokoena" /></Field>
          <Field label="Area / Location"><input style={inputStyle} value={newConnector.area} onChange={e => setNewConnector(p => ({...p, area: e.target.value}))} placeholder="e.g. Sandton, Gauteng" /></Field>
          <Field label="Type">
            <select style={selectStyle} value={newConnector.type} onChange={e => setNewConnector(p => ({...p, type: e.target.value}))}>
              {TYPES.map(t => <option key={t}>{t}</option>)}
            </select>
          </Field>
          <Field label="Initial Tag">
            <select style={selectStyle} value={newConnector.tag} onChange={e => setNewConnector(p => ({...p, tag: e.target.value}))}>
              {TAGS.map(t => <option key={t}>{t}</option>)}
            </select>
          </Field>
          <div style={{ display: "flex", gap: 10, marginTop: 8 }}>
            <button onClick={addConnector} style={{ flex: 1, background: "#1A6FFF", border: "none", color: "#fff", padding: "12px", borderRadius: 6, cursor: "pointer", fontWeight: 600, fontSize: 13, fontFamily: "inherit" }}>Add to Network</button>
            <button onClick={() => setShowAddConnector(false)} style={{ flex: 1, background: "transparent", border: "1px solid rgba(200,214,229,0.15)", color: "#6B7A94", padding: "12px", borderRadius: 6, cursor: "pointer", fontFamily: "inherit" }}>Cancel</button>
          </div>
        </Modal>
      )}

      {/* ── EDIT CONNECTOR MODAL ── */}
      {editConnector && (
        <Modal title="Edit Connector" onClose={() => setEditConnector(null)}>
          <Field label="Full Name"><input style={inputStyle} value={editConnector.name} onChange={e => setEditConnector(p => ({...p, name: e.target.value}))} /></Field>
          <Field label="Area"><input style={inputStyle} value={editConnector.area} onChange={e => setEditConnector(p => ({...p, area: e.target.value}))} /></Field>
          <Field label="Tag">
            <select style={selectStyle} value={editConnector.tag} onChange={e => setEditConnector(p => ({...p, tag: e.target.value}))}>
              {TAGS.map(t => <option key={t}>{t}</option>)}
            </select>
          </Field>
          <Field label="Referrals Made"><input style={inputStyle} type="number" value={editConnector.referrals} onChange={e => setEditConnector(p => ({...p, referrals: parseInt(e.target.value)||0}))} /></Field>
          <Field label="Converted"><input style={inputStyle} type="number" value={editConnector.converted} onChange={e => setEditConnector(p => ({...p, converted: parseInt(e.target.value)||0}))} /></Field>
          <Field label="Total Revenue Earned (R)"><input style={inputStyle} type="number" value={editConnector.revenue} onChange={e => setEditConnector(p => ({...p, revenue: parseInt(e.target.value)||0}))} /></Field>
          <div style={{ display: "flex", alignItems: "center", gap: 10, marginBottom: 16 }}>
            <input type="checkbox" id="agreementSigned" checked={!!editConnector.agreementSigned} onChange={e => setEditConnector(p => ({...p, agreementSigned: e.target.checked, agreementSignedDate: e.target.checked ? (p.agreementSignedDate || new Date().toISOString().slice(0,10)) : null}))} style={{ width: 16, height: 16 }} />
            <label htmlFor="agreementSigned" style={{ fontSize: 13, color: "#C8D6E5" }}>Partner agreement signed{editConnector.agreementSignedDate ? ` (${editConnector.agreementSignedDate})` : ""}</label>
          </div>
          <div style={{ display: "flex", gap: 10, marginTop: 8 }}>
            <button onClick={saveConnector} style={{ flex: 1, background: "#1A6FFF", border: "none", color: "#fff", padding: "12px", borderRadius: 6, cursor: "pointer", fontWeight: 600, fontSize: 13, fontFamily: "inherit" }}>Save Changes</button>
            <button onClick={() => deleteConnector(editConnector.id)} style={{ background: "rgba(239,68,68,0.1)", border: "1px solid rgba(239,68,68,0.25)", color: "#F87171", padding: "12px 20px", borderRadius: 6, cursor: "pointer", fontFamily: "inherit", fontSize: 13 }}>Delete</button>
          </div>
        </Modal>
      )}

      {/* ── ADD LEAD MODAL ── */}
      {showAddLead && (
        <Modal title="Add Lead / Opportunity" onClose={() => setShowAddLead(false)}>
          <Field label="Need / Service *"><input style={inputStyle} value={newLead.need} onChange={e => setNewLead(p => ({...p, need: e.target.value}))} placeholder="e.g. Artificial grass installation" /></Field>
          <Field label="Area"><input style={inputStyle} value={newLead.area} onChange={e => setNewLead(p => ({...p, area: e.target.value}))} placeholder="e.g. Sandton" /></Field>
          <Field label="Size / Scope"><input style={inputStyle} value={newLead.size} onChange={e => setNewLead(p => ({...p, size: e.target.value}))} placeholder="e.g. 80m²" /></Field>
          <Field label="Budget Signal">
            <select style={selectStyle} value={newLead.budget} onChange={e => setNewLead(p => ({...p, budget: e.target.value}))}>
              {["Low","Medium","High"].map(b => <option key={b}>{b}</option>)}
            </select>
          </Field>
          <Field label="Status">
            <select style={selectStyle} value={newLead.status} onChange={e => setNewLead(p => ({...p, status: e.target.value}))}>
              {LEAD_STATUSES.map(s => <option key={s}>{s}</option>)}
            </select>
          </Field>
          <Field label="Connector (who referred this)">
            <select style={selectStyle} value={newLead.connectorId} onChange={e => setNewLead(p => ({...p, connectorId: e.target.value}))}>
              <option value="">— None yet —</option>
              {data.connectors.map(c => <option key={c.id} value={c.id}>{c.name} ({c.area})</option>)}
            </select>
          </Field>
          <Field label="Reward Paid (R)"><input style={inputStyle} type="number" value={newLead.reward} onChange={e => setNewLead(p => ({...p, reward: e.target.value}))} placeholder="0" /></Field>
          <Field label="Notes"><input style={inputStyle} value={newLead.notes} onChange={e => setNewLead(p => ({...p, notes: e.target.value}))} placeholder="Any context..." /></Field>
          <div style={{ display: "flex", gap: 10, marginTop: 8 }}>
            <button onClick={addLead} style={{ flex: 1, background: "#1A6FFF", border: "none", color: "#fff", padding: "12px", borderRadius: 6, cursor: "pointer", fontWeight: 600, fontSize: 13, fontFamily: "inherit" }}>Add Lead</button>
            <button onClick={() => setShowAddLead(false)} style={{ flex: 1, background: "transparent", border: "1px solid rgba(200,214,229,0.15)", color: "#6B7A94", padding: "12px", borderRadius: 6, cursor: "pointer", fontFamily: "inherit" }}>Cancel</button>
          </div>
        </Modal>
      )}

      {/* ── EDIT LEAD MODAL ── */}
      {editLead && (
        <Modal title="Edit Lead" onClose={() => setEditLead(null)}>
          <Field label="Need / Service"><input style={inputStyle} value={editLead.need} onChange={e => setEditLead(p => ({...p, need: e.target.value}))} /></Field>
          <Field label="Area"><input style={inputStyle} value={editLead.area} onChange={e => setEditLead(p => ({...p, area: e.target.value}))} /></Field>
          <Field label="Status">
            <select style={selectStyle} value={editLead.status} onChange={e => setEditLead(p => ({...p, status: e.target.value}))}>
              {LEAD_STATUSES.map(s => <option key={s}>{s}</option>)}
            </select>
          </Field>
          <Field label="Connector">
            <select style={selectStyle} value={editLead.connectorId || ""} onChange={e => setEditLead(p => ({...p, connectorId: e.target.value ? parseInt(e.target.value) : null}))}>
              <option value="">— None —</option>
              {data.connectors.map(c => <option key={c.id} value={c.id}>{c.name}</option>)}
            </select>
          </Field>
          <Field label="Reward Paid (R)"><input style={inputStyle} type="number" value={editLead.reward} onChange={e => setEditLead(p => ({...p, reward: e.target.value}))} /></Field>
          <Field label="Notes"><input style={inputStyle} value={editLead.notes} onChange={e => setEditLead(p => ({...p, notes: e.target.value}))} /></Field>
          <div style={{ display: "flex", gap: 10, marginTop: 8 }}>
            <button onClick={saveLead} style={{ flex: 1, background: "#1A6FFF", border: "none", color: "#fff", padding: "12px", borderRadius: 6, cursor: "pointer", fontWeight: 600, fontSize: 13, fontFamily: "inherit" }}>Save Changes</button>
            <button onClick={() => deleteLead(editLead.id)} style={{ background: "rgba(239,68,68,0.1)", border: "1px solid rgba(239,68,68,0.25)", color: "#F87171", padding: "12px 20px", borderRadius: 6, cursor: "pointer", fontFamily: "inherit", fontSize: 13 }}>Delete</button>
          </div>
        </Modal>
      )}

    </div>
  );
}
